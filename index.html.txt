<script>
(function(){
  const room=document.getElementById('room'), key=document.getElementById('key'),
        door=document.getElementById('door'), char1=document.getElementById('char1'),
        char2=document.getElementById('char2'), hint=document.getElementById('hint'),
        msg=document.getElementById('msg'), win=document.getElementById('win'),
        flowers=document.getElementById('flowers');

  let keyState='in_hand';
  let doorState='locked';
  let char2State='hidden';
  let char1State='sad';
  let kissDone=false;

  function placeKeyInHand(){
    const c=char1.getBoundingClientRect(), r=room.getBoundingClientRect();
    key.style.left=(c.left-r.left+c.width*0.72)+'px';
    key.style.top =(c.top -r.top +c.height*0.62)+'px';
  }
  window.addEventListener('load',placeKeyInHand);
  window.addEventListener('resize',()=>{ if(keyState==='in_hand') placeKeyInHand(); });
  placeKeyInHand();

  const setHint=t=>hint.textContent=t;
  const showMsg=()=>{ msg.classList.add('show'); setTimeout(()=>msg.classList.remove('show'),1600); };

  function openDoor(){
    if(doorState!=='locked') return;
    doorState='open';
    room.classList.add('opened');
    char2State='visible';
    char2.classList.add('visible');
    setHint('Дверь открылась! А там… кто-то ждал тебя 💫');
  }

  function sendKiss(){
    const c1=char1.getBoundingClientRect(), r=room.getBoundingClientRect();
    for(let i=0;i<6;i++){
      const h=document.createElement('div'); h.className='heart'; h.textContent='💋';
      h.style.left=(c1.left-r.left+c1.width*0.85)+'px'; h.style.top=(c1.top-r.top+c1.height*0.35)+'px';
      h.style.setProperty('--dx',(Math.random()*100+20)+'px'); h.style.animationDelay=(i*0.15)+'s';
      room.appendChild(h); setTimeout(()=>h.remove(),1800+i*150);
    }
  }

  function doKiss(){
    if(kissDone) return;
    kissDone=true;
    char1State='happy';
    char1.className='happy';
    const c2=char2.getBoundingClientRect(), c1=char1.getBoundingClientRect(), r=room.getBoundingClientRect();
    const targetLeft = (c2.left-r.left) - c1.width - 10;
    char1.style.left = targetLeft+'px';
    char1.style.transform='scaleX(1)';
    setHint('Он идёт к ней… 💕');
    setTimeout(()=>{
      sendKiss();
      char2.className='visible kissed';
      setHint('Поцелуй! 💋💋💋');
    },1200);
    setTimeout(()=>{
      // Мальчик дарит цветы
      const c2=char2.getBoundingClientRect(), r=room.getBoundingClientRect();
      flowers.style.left=(c2.left-r.left+c2.width*0.3)+'px';
      flowers.style.top=(c2.top-r.top+c2.height*0.4)+'px';
      flowers.classList.add('visible');
      setHint('Он дарит ей цветы! 🌸💐🌺');
      setTimeout(()=>win.classList.add('show'),1200);
    },2400);
  }

  key.addEventListener('click',function(e){
    e.stopPropagation();
    if(keyState!=='in_hand') return;
    keyState='flying';
    key.classList.remove('idle');
    const lk=document.getElementById('lock').getBoundingClientRect(), r=room.getBoundingClientRect();
    key.style.transition='top .9s ease-in,left .9s ease-in,transform .9s ease-in';
    key.style.left=(lk.left-r.left+lk.width/2-30)+'px';
    key.style.top =(lk.top -r.top +lk.height/2-13)+'px';
    key.style.transform='rotate(90deg) scale(.9)';
    setHint('Ключ летит к замку… проворачивается…');
    setTimeout(()=>{ key.style.transform='rotate(360deg) scale(.7)'; },900);
    setTimeout(()=>{ key.classList.add('gone'); keyState='gone'; },1300);
    setTimeout(openDoor,1500);
  });

  door.addEventListener('click',function(e){
    e.stopPropagation();
    if(doorState==='open'){ setHint('Дверь уже открыта!'); return; }
    if(keyState==='gone'){ openDoor(); return; }
    door.classList.remove('shake'); void door.offsetWidth; door.classList.add('shake');
    setHint('Заперто! Сначала возьми ключ у девочки.');
  });

  door.addEventListener('contextmenu',function(e){
    e.preventDefault(); e.stopPropagation();
    showMsg();
    setHint('«И я сейчас уйду»… 🚪💨');
  });

  char1.addEventListener('click',function(e){
    e.stopPropagation();
    if(doorState==='locked'){
      char1State='angry';
      char1.className='angry turn';
      setHint('Она нахмурилась и отвернулась от тебя 😠');
      setTimeout(()=>{ char1.className='sad turn'; char1State='sad'; },1600);
    }else{
      doKiss();
    }
  });

  char2.addEventListener('click',function(e){
    e.stopPropagation();
    if(char2State==='visible' && doorState==='open'){
      setHint('Нажми на девочку слева, чтобы она подошла к нему 💕');
    }
  });
})();
